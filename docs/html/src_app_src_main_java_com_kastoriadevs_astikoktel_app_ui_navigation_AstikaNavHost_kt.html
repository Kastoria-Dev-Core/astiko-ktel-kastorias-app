52/**
53 * Bottom tabs. Screens opened from a tab stay part of it, so the highlighted tab is
54 * tracked from the back stack rather than from the screen shown: a map opened with
55 * "Show on map" from a stop still belongs to Home.
56 */
57private enum class Tab(@StringRes val label: Int, val route: Any, val isRoot: (NavDestination) -> Boolean) {
58 HOME(R.string.tab_home, HomeRoute, { it.hasRoute<HomeRoute>() }),
59 LINES(R.string.tab_lines, LinesRoute, { it.hasRoute<LinesRoute>() }),
60 MAP(R.string.tab_map, MapRoute(), { it.hasRoute<MapRoute>() }),
61}

70private fun NavHostController.switchTab(tab: Tab) {
71 navigate(tab.route) {
72 popUpTo(graph.findStartDestination().id) { saveState = true }

78/** Back to the first screen of [tab], like tapping the tab you're already on in most apps */
79private fun NavHostController.popToTabRoot(tab: Tab) {
80 val startId = graph.findStartDestination().id
81 if (tab == Tab.HOME) {
82 popBackStack(startId, inclusive = false)
83 return
84 }
85 // Other tabs sit right above Home: [Home, tab root, details...]
86 while (previousBackStackEntry != null && previousBackStackEntry?.destination?.id != startId) {
87 popBackStack()
88 }
89}

91@Composable
92fun AstikaNavHost(appVm: AppViewModel) {
93 val nav = rememberNavController()
94 val backStack by nav.currentBackStackEntryAsState()
95 var currentTab by rememberSaveable { mutableStateOf(Tab.HOME) }
96
97 // A tab root reached by back (e.g. back from Lines lands on Home) moves the highlight.
98 // A tab root opened inside another tab (the map from "Show on map") doesn't.
99 LaunchedEffect(backStack) {
100 val destination = backStack?.destination ?: return@LaunchedEffect
101 val tab = Tab.entries.firstOrNull { it.isRoot(destination) } ?: return@LaunchedEffect
102 val startId = nav.graph.findStartDestination().id
103 if (tab == Tab.HOME || nav.previousBackStackEntry?.destination?.id == startId) currentTab = tab
104 }
