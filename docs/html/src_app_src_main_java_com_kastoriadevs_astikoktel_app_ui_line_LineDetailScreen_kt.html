253/** "Every 20′ · 07:00 to 20:15", or just the first and last time when the gaps differ */
254@Composable
255private fun serviceSummary(deps: List<Int>): String? {
256 if (deps.isEmpty()) return null
257 val span = stringResource(R.string.first_last, Schedule.formatMinutes(deps.first()), Schedule.formatMinutes(deps.last()))
258 val gaps = deps.zipWithNext { a, b -> b - a }.toSet()
259 return if (gaps.size == 1) stringResource(R.string.every_min, gaps.first()) + " · " + span else span
260}

315/** A bus on a route line: a dark rounded square with a white outline and a simple bus inside */
316internal fun DrawScope.busMarker(center: Offset, fill: Color, outline: Color) {
317 val half = 13.dp.toPx()
318 val ring = 2.dp.toPx()
319 drawRoundRect(outline, Offset(center.x - half - ring, center.y - half - ring), Size((half + ring) * 2, (half + ring) * 2), CornerRadius(half))
320 drawRoundRect(fill, Offset(center.x - half, center.y - half), Size(half * 2, half * 2), CornerRadius(half))
321 // Body, windscreen, wheels
322 val w = 6.5.dp.toPx()
323 val h = 7.5.dp.toPx()
324 drawRoundRect(outline, Offset(center.x - w, center.y - h), Size(w * 2, h * 1.7f), CornerRadius(2.dp.toPx()))
325 drawRect(fill, Offset(center.x - w + 1.5.dp.toPx(), center.y - h + 1.5.dp.toPx()), Size(w * 2 - 3.dp.toPx(), h * 0.75f))
326 drawRect(outline, Offset(center.x - w + 1.dp.toPx(), center.y + h * 0.7f), Size(2.5.dp.toPx(), 2.5.dp.toPx()))
327 drawRect(outline, Offset(center.x + w - 3.5.dp.toPx(), center.y + h * 0.7f), Size(2.5.dp.toPx(), 2.5.dp.toPx()))
328}
