25/**
26 * Works out where each bus should be from the timetable: a trip is on the road between leaving its
27 * first stop and reaching its last, and between two stops it moves along the line's street path
28 * at the pace the stop times give.
29 */
30class BusEstimator(private val data: TransitData, private val schedule: Schedule) {

51 /** The buses on the road at [now] */
52 fun estimate(now: LocalDateTime): List<EstimatedBus> {
53 val nowSeconds = now.toLocalTime().toSecondOfDay().toDouble()
54 val date = now.toLocalDate()
55 return tracks.flatMap { track ->
56 val tripSeconds = track.stopTime.last()
57 schedule.departuresFor(track.line.id, track.direction.id, date).mapNotNull { departure ->
58 val elapsed = nowSeconds - departure * 60.0
59 if (elapsed < 0 || elapsed >= tripSeconds) null else position(track, departure, elapsed)
60 }
61 }
62 }

64 private fun position(track: Track, departure: Int, elapsed: Double): EstimatedBus? {
65 val times = track.stopTime
66 // The stretch between two stops the bus is on. Stretches of no time (same minute) are skipped.
67 val i = (0 until times.size - 1).firstOrNull { elapsed >= times[it] && elapsed < times[it + 1] } ?: return null
68 val fraction = (elapsed - times[i]) / (times[i + 1] - times[i])
69 val distance = track.stopDistance[i] + fraction * (track.stopDistance[i + 1] - track.stopDistance[i])
70 val (lat, lng, bearing) = pointAt(track, distance)
71 return EstimatedBus(track.line, track.direction, departure, lat, lng, bearing, track.stopIds[i + 1])
72 }

74 /** The point [distance] metres along the track, and the bearing of the path there */
75 private fun pointAt(track: Track, distance: Double): Triple<Double, Double, Double> {
76 val cumulative = track.cumulative
77 var segment = cumulative.indexOfLast { it <= distance }.coerceIn(0, cumulative.size - 2)
78 // Step past zero-length segments so the bearing has a direction
79 while (segment < cumulative.size - 2 && cumulative[segment + 1] == cumulative[segment]) segment++
80 val length = cumulative[segment + 1] - cumulative[segment]
81 val t = if (length == 0.0) 0.0 else ((distance - cumulative[segment]) / length).coerceIn(0.0, 1.0)
82 val (aLat, aLng) = track.points[segment]
83 val (bLat, bLng) = track.points[segment + 1]
84 val (ax, ay) = track.xy[segment]
85 val (bx, by) = track.xy[segment + 1]
86 val bearing = (Math.toDegrees(atan2(bx - ax, by - ay)) + 360) % 360
87 return Triple(aLat + t * (bLat - aLat), aLng + t * (bLng - aLng), bearing)
88 }

90 private fun track(line: Line, direction: Direction): Track? {
91 val stops = direction.stops.mapNotNull { s -> data.stop(s.stop)?.let { it to s.offset } }
92 if (stops.size < 2) return null
93 val points = data.directionPath(line, direction.id).ifEmpty { stops.map { it.first.lat to it.first.lng } }
94 if (points.size < 2) return null
95
96 val lat0 = points.first().first
97 val kx = 111_320.0 * cos(Math.toRadians(lat0))
98 val ky = 110_540.0
99 fun xy(lat: Double, lng: Double) = lng * kx to lat * ky
100 val xy = points.map { (lat, lng) -> xy(lat, lng) }
101 val cumulative = DoubleArray(points.size)
102 for (k in 1 until xy.size) {
103 cumulative[k] = cumulative[k - 1] + hypot(xy[k].first - xy[k - 1].first, xy[k].second - xy[k - 1].second)
104 }
105
106 // Each stop's place along the path. Always searching forward from the previous stop,
107 // so a path that passes the same street twice puts each stop on the right pass.
108 val stopDistance = DoubleArray(stops.size)
109 var segment = 0
110 stops.forEachIndexed { index, (stop, _) ->
111 val (sx, sy) = xy(stop.lat, stop.lng)
112 val reach = cumulative[segment] + SEARCH_AHEAD_METERS
113 var best = Double.MAX_VALUE
114 var bestSegment = segment
115 var bestAlong = cumulative[segment]
116 var j = segment
117 while (j < xy.size - 1 && cumulative[j] <= reach) {
118 val (d, along) = project(sx, sy, xy[j], xy[j + 1])
119 if (d < best) {
120 best = d
121 bestSegment = j
122 bestAlong = cumulative[j] + along
123 }
124 j++
125 }
126 segment = bestSegment
127 stopDistance[index] = if (index == 0) bestAlong else maxOf(bestAlong, stopDistance[index - 1])
128 }
129
130 return Track(line, direction, points, xy, cumulative, stops.map { it.first.id }, stopDistance, smoothTimes(stops.map { it.second }, stopDistance))
131 }

134 /** How far along the path to look for the next stop */

137 /** Distance from point (px, py) to segment ab, and how far along ab the closest point is, in metres */
138 private fun project(px: Double, py: Double, a: Pair<Double, Double>, b: Pair<Double, Double>): Pair<Double, Double> {
139 val dx = b.first - a.first
140 val dy = b.second - a.second
141 val length2 = dx * dx + dy * dy
142 val t = if (length2 == 0.0) 0.0 else (((px - a.first) * dx + (py - a.second) * dy) / length2).coerceIn(0.0, 1.0)
143 return hypot(px - a.first - t * dx, py - a.second - t * dy) to t * kotlin.math.sqrt(length2)
144 }

146 /**
147 * Stop times in seconds. The timetable gives whole minutes, so several stops in a row can share
148 * a minute and the bus would jump between them. Those in-between stops get times spread by
149 * distance between the last stop before them and the next stop with a later minute.
150 */
151 fun smoothTimes(offsetMinutes: List<Int>, stopDistance: DoubleArray): DoubleArray {
152 val seconds = DoubleArray(offsetMinutes.size) { offsetMinutes[it] * 60.0 }
153 var i = 0
154 while (i < offsetMinutes.size) {
155 var next = i + 1
156 while (next < offsetMinutes.size && offsetMinutes[next] == offsetMinutes[i]) next++
157 if (next < offsetMinutes.size && next > i + 1) {
158 val span = stopDistance[next] - stopDistance[i]
159 for (m in i + 1 until next) {
160 val share = if (span > 0) (stopDistance[m] - stopDistance[i]) / span else (m - i).toDouble() / (next - i)
161 seconds[m] = seconds[i] + share * (seconds[next] - seconds[i])
162 }
163 }
164 i = next
165 }
166 return seconds
167 }
168 }
