21/** One-shot current location through the fused provider */
22class LocationClient(private val context: Context) {
23
24 private val fused = LocationServices.getFusedLocationProviderClient(context)
25
26 fun hasPermission(): Boolean =
27 ContextCompat.checkSelfPermission(context, Manifest.permission.ACCESS_FINE_LOCATION) ==
28 PackageManager.PERMISSION_GRANTED ||
29 ContextCompat.checkSelfPermission(context, Manifest.permission.ACCESS_COARSE_LOCATION) ==
30 PackageManager.PERMISSION_GRANTED
31
32 private fun locationEnabled(): Boolean {
33 val lm = context.getSystemService(Context.LOCATION_SERVICE) as LocationManager
34 return lm.isProviderEnabled(LocationManager.GPS_PROVIDER) ||
35 lm.isProviderEnabled(LocationManager.NETWORK_PROVIDER)
36 }
37
38 /**
39 * Call only after the permission is granted. Indoors a fresh GPS fix can take very
40 * long or never come, so after [FIX_TIMEOUT_MS] it settles for the last known position.
41 */
42 suspend fun current(): Coords {
43 if (!locationEnabled()) throw LocationDisabledException()
44 return withTimeoutOrNull(FIX_TIMEOUT_MS) { freshFix() }
45 ?: lastKnown()
46 ?: throw IllegalStateException("No location")
47 }
48
49 @SuppressLint("MissingPermission")
50 private suspend fun freshFix(): Coords {
51 val cancel = CancellationTokenSource()
52 return suspendCancellableCoroutine { cont ->
53 cont.invokeOnCancellation { cancel.cancel() }
54 fused.getCurrentLocation(Priority.PRIORITY_HIGH_ACCURACY, cancel.token)
55 .addOnSuccessListener { loc ->
56 if (loc != null) cont.resume(Coords(loc.latitude, loc.longitude, loc.accuracy))
57 else cont.resumeWithException(IllegalStateException("No location"))
58 }
59 .addOnFailureListener { cont.resumeWithException(it) }
60 }
61 }
62
63 @SuppressLint("MissingPermission")
64 private suspend fun lastKnown(): Coords? = suspendCancellableCoroutine { cont ->
65 fused.lastLocation
66 .addOnSuccessListener { loc ->
67 // An old position can be anywhere the phone was earlier, so only a recent one will do
68 val ageNanos = loc?.let { SystemClock.elapsedRealtimeNanos() - it.elapsedRealtimeNanos }
69 val recent = ageNanos != null && ageNanos <= MAX_LAST_KNOWN_AGE_NANOS
70 cont.resume(loc?.takeIf { recent }?.let { Coords(it.latitude, it.longitude, it.accuracy) })
71 }
72 .addOnFailureListener { cont.resume(null) }
73 }
74
75 private companion object {
76 const val FIX_TIMEOUT_MS = 10_000L
77 const val MAX_LAST_KNOWN_AGE_NANOS = 2 * 60 * 1_000_000_000L
78 }
79}
