56/**
57 * Finds the buses that go from near the user to a [Destination], using only the
58 * direction stop lists and today's timetables.
59 */
60object JourneyPlanner {
61
62 /** Stops up to this far are an easy walk: we look here first */
63 const val MAX_WALK_METERS = 1_000.0
64
65 /** If no stop within [MAX_WALK_METERS] goes there, we look this far for one that does */
66 const val MAX_SEARCH_METERS = 3_000.0
67
68 /** Closer than this to a destination stop: you are already there */
69 const val ALREADY_THERE_METERS = 300.0
70
71 /** Rides from stops this much further than the closest useful stop are shown too */
72 private const val EXTRA_WALK_METERS = 300.0
73
74 fun plan(
75 data: TransitData,
76 schedule: Schedule,
77 destination: Destination,
78 lat: Double,
79 lng: Double,
80 now: LocalDateTime,
81 ): JourneyPlan? {
82 val nearby = data.stops
83 .map { NearbyStop(it, Geo.distanceMeters(lat, lng, it.lat, it.lng)) }
84 .sortedBy { it.distanceMeters }
85 val nearest = nearby.firstOrNull() ?: return null
86 // Outside Kastoria no stop is a walk away, however close the "closest" one is
87 if (nearest.distanceMeters > MAX_SEARCH_METERS) {
88 return JourneyPlan(JourneyStatus.TooFar, nearest, null, emptyList())
89 }
90 if (isAlreadyThere(nearby, destination)) {
91 return JourneyPlan(JourneyStatus.AlreadyThere, nearest, null, emptyList())
92 }
93 // A direction with no timetable never has a bus to show, so it can't take you there
94 val timed = data.timetables.filter { it.departures.isNotEmpty() }.map { it.line to it.direction }.toSet()
95 val lines = data.lines.map { line -> line.copy(directions = line.directions.filter { (line.id to it.id) in timed }) }
96 // First the stops within an easy walk; if none of them goes there, look further
97 val rides = nearestRides(findRides(stopsWithin(nearby, MAX_WALK_METERS), lines, destination))
98 .ifEmpty { nearestRides(findRides(stopsWithin(nearby, MAX_SEARCH_METERS), lines, destination)) }
99 if (rides.isEmpty()) return JourneyPlan(JourneyStatus.NoRoute, nearest, null, emptyList())
100
101 val options = options(rides, schedule, now)
102 // The closest stop among the buses we show (a stop whose bus already left today
103 // would be confusing), or among all rides when no bus is left today
104 val useful = (options.map { it.ride }.ifEmpty { rides })
105 .minBy { it.board.distanceMeters }.board.takeIf { it.stop.id != nearest.stop.id }
106 return JourneyPlan(
107 status = if (options.isEmpty()) JourneyStatus.NoMoreBuses else JourneyStatus.Ready,
108 nearestStop = nearest,
109 usefulStop = useful,
110 options = options,
111 )
112 }

114 /** The stops up to [meters] away, closest first. The closest stop is always included. */
115 fun stopsWithin(nearbySorted: List<NearbyStop>, meters: Double): List<NearbyStop> =
116 nearbySorted.filterIndexed { index, stop -> index == 0 || stop.distanceMeters <= meters }
117
118 /**
119 * Keeps the rides worth walking to: those from stops within [MAX_WALK_METERS] or, when
120 * the closest stop that goes there is further away, from that stop and the ones up to
121 * [EXTRA_WALK_METERS] further (another line may leave sooner from a stop just beyond it).
122 */
123 fun nearestRides(rides: List<Ride>): List<Ride> {
124 val closest = rides.minOfOrNull { it.board.distanceMeters } ?: return emptyList()
125 val limit = maxOf(MAX_WALK_METERS, closest + EXTRA_WALK_METERS)
126 return rides.filter { it.board.distanceMeters <= limit }
127 }
128
129 fun isAlreadyThere(nearbySorted: List<NearbyStop>, destination: Destination): Boolean =
130 nearbySorted.any { it.stop.id in destination.stopIds && it.distanceMeters <= ALREADY_THERE_METERS }
131
132 /**
133 * For each direction, the closest boarding stop from which it LATER reaches a destination
134 * stop (a bus only drives forward along its stop list, so a stop after the destination is
135 * the wrong way). [boardingStops] must be closest first.
136 */
137 fun findRides(boardingStops: List<NearbyStop>, lines: List<Line>, destination: Destination): List<Ride> =
138 lines.flatMap { line ->
139 line.directions.mapNotNull { direction ->
140 val ids = direction.stops.map { it.stop }
141 boardingStops.firstNotNullOfOrNull { board ->
142 if (board.stop.id in destination.stopIds) return@firstNotNullOfOrNull null
143 // A stop can appear twice in a direction (rare): try every pass
144 ids.indices.filter { ids[it] == board.stop.id }.firstNotNullOfOrNull { boardIndex ->
145 (boardIndex + 1 until ids.size).firstOrNull { ids[it] in destination.stopIds }?.let {
146 Ride(line, direction, board, direction.stops[boardIndex].offset, ids[it])
147 }
148 }
149 }
150 }
151 }

153 /**
154 * The next catchable bus of every ride today, soonest first. A bus that leaves before you
155 * could walk to the stop is skipped.
156 */
157 fun options(rides: List<Ride>, schedule: Schedule, now: LocalDateTime): List<JourneyOption> {
158 val nowMin = Schedule.minutesOfDay(now)
159 return rides.mapNotNull { ride ->
160 val catchable = schedule.departuresFor(ride.line.id, ride.direction.id, now.toLocalDate())
161 .map { it + ride.boardOffset }
162 .filter { it - nowMin >= ride.board.walkMinutes }
163 catchable.firstOrNull()?.let { JourneyOption(ride, it, it - nowMin, catchable.getOrNull(1)) }
164 }.sortedWith(compareBy<JourneyOption> { it.minutesUntil }.thenBy { it.ride.board.distanceMeters })
165 }
