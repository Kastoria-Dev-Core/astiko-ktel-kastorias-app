17 /**
18 * Stops a bus leaves from on at least one day. Leaves out stops that are only the
19 * last stop of a direction (buses arrive there, like Αφετηρία 2) and stops served
20 * only by lines that have no timetable.
21 */
22 val departureStopIds: Set<String> by lazy {
23 data.lines.flatMap { line ->
24 line.directions
25 .filter { d -> data.timetables.any { it.line == line.id && it.direction == d.id && it.departures.isNotEmpty() } }
26 .flatMap { d -> d.stops.dropLast(1).map { it.stop } }
27 }.toSet()
28 }

31 fun serviceDayFor(date: LocalDate): ServiceDay = when {
32 isHoliday(date) -> ServiceDay.SUNDAY
33 date.dayOfWeek == DayOfWeek.SUNDAY -> ServiceDay.SUNDAY
34 date.dayOfWeek == DayOfWeek.SATURDAY -> ServiceDay.SATURDAY
35 else -> ServiceDay.WEEKDAY

39 val key = mmdd(date)

49 /** Departures from the first stop, minutes since midnight, sorted. Falls back to the winter timetable. */
50 fun departuresFor(lineId: String, directionId: String, service: ServiceDay, season: Season): List<Int> {
51 val tt = data.timetables.firstOrNull {
52 it.line == lineId && it.direction == directionId && it.service == service && it.season == season
53 } ?: data.timetables.firstOrNull {
54 it.line == lineId && it.direction == directionId && it.service == service && it.season == Season.WINTER
55 }
56 return tt?.departures?.map(::toMinutes)?.sorted().orEmpty()
57 }

59 fun departuresFor(lineId: String, directionId: String, date: LocalDate): List<Int> =

62 /** Upcoming departures from a stop: the rest of today, then tomorrow's first ones */
63 fun nextDepartures(stopId: String, now: LocalDateTime, limit: Int): List<Departure> {
64 val nowMin = minutesOfDay(now)
65 val today = departuresAtStopOn(stopId, now.toLocalDate(), dayOffset = 0)
66 .filter { it.minutes >= nowMin }
67 .sortedBy { it.minutes }
68 if (today.size >= limit) return today.take(limit)
69 val tomorrow = departuresAtStopOn(stopId, now.toLocalDate().plusDays(1), dayOffset = 1)
70 .sortedBy { it.minutes }
71 return (today + tomorrow).take(limit)
72 }

74 /** The bus that left a stop most recently today, if any left in the last [withinMinutes] */
75 fun previousDeparture(stopId: String, now: LocalDateTime, withinMinutes: Int = 30): Departure? {
76 val nowMin = minutesOfDay(now)
77 return departuresAtStopOn(stopId, now.toLocalDate(), dayOffset = 0)
78 .filter { it.minutes < nowMin && it.minutes >= nowMin - withinMinutes }
79 .maxByOrNull { it.minutes }

92 /**
93 * Where the buses on the road are along a direction right now, from the timetable, counted in
94 * stops from the first: 2.5 is halfway between the third and the fourth stop.
95 */
96 fun positionsOnRoute(line: Line, direction: Direction, now: LocalDateTime): List<Double> {
97 val offsets = direction.stops.map { it.offset }
98 if (offsets.size < 2) return emptyList()
99 val nowMin = now.toLocalTime().toSecondOfDay() / 60.0
100 val total = offsets.last()
101 return departuresFor(line.id, direction.id, now.toLocalDate()).mapNotNull { dep ->
102 val elapsed = nowMin - dep
103 if (elapsed < 0 || elapsed >= total) return@mapNotNull null
104 // The last stretch that has started; stops in the same minute are passed together
105 val i = offsets.indices.lastOrNull { it < offsets.lastIndex && offsets[it] <= elapsed && offsets[it + 1] > offsets[it] }
106 ?: return@mapNotNull null
107 i + (elapsed - offsets[i]) / (offsets[i + 1] - offsets[i])
108 }
109 }

117 /** All departures from a stop on a date. The last stop of a direction is skipped (buses arrive, they don't leave). */
118 private fun departuresAtStopOn(stopId: String, date: LocalDate, dayOffset: Int): List<Departure> {
119 val service = serviceDayFor(date)
120 val season = seasonFor(date)
121 val out = mutableListOf<Departure>()
122 for (line in data.lines) {
123 for (direction in line.directions) {
124 val idx = direction.stops.indexOfFirst { it.stop == stopId }
125 if (idx == -1 || idx == direction.stops.lastIndex) continue
126 val offset = direction.stops[idx].offset
127 val deps = departuresFor(line.id, direction.id, service, season)
128 deps.forEachIndexed { i, dep ->
129 out += Departure(line, direction, dep + offset, dayOffset, last = i == deps.lastIndex)
130 }
131 }
132 }
133 return out

153 /** Orthodox Easter Sunday (Gregorian date), valid 1900-2099 */
154 fun orthodoxEaster(year: Int): LocalDate {
155 val a = year % 4
156 val b = year % 7
157 val c = year % 19
158 val d = (19 * c + 15) % 30
159 val e = (2 * a + 4 * b - d + 34) % 7
160 val month = (d + e + 114) / 31
161 val day = ((d + e + 114) % 31) + 1
162 return LocalDate.of(year, month, day).plusDays(13)
163 }

165 /** Clean Monday, Good Friday, Easter Monday, Holy Spirit Monday */
166 fun movableHolidays(year: Int): List<String> {
