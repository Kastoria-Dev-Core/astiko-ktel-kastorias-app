955/**
956 * All the stops in one overlay: pins from street level up, small dots further out where 209 pins would
957 * cover the map. The open stop always gets its big pin, drawn last so it sits on top.
958 */
959private class StopsOverlay(
960 density: Float,
961 private val pin: Bitmap,
962 private val openPin: Bitmap,
963 private val onTap: (String) -> Unit,
964) : Overlay() {
965 var stops: List<Pair<String, GeoPoint>> = emptyList()
966 var selectedId: String? = null
967
968 private val point = Point()
969 private val dotRadius = 3f * density
970 private val dotRing = 1.25f * density
971 private val tapRadius = 22f * density
972 private val dotFill = Paint(Paint.ANTI_ALIAS_FLAG).apply { color = Lake.toArgb() }
973 private val dotStroke = Paint(Paint.ANTI_ALIAS_FLAG).apply { color = 0xFFFFFFFF.toInt() }
974
975 private fun showPins(map: MapView) = map.zoomLevelDouble >= PIN_ZOOM
976
977 /** Far out the routes say enough; the stops would only be a string of beads along them */
978 private fun showStops(map: MapView) = map.zoomLevelDouble >= DOT_ZOOM
979
980 override fun draw(canvas: Canvas, map: MapView, shadow: Boolean) {
981 if (shadow) return
982 val projection = map.projection
983 val pins = showPins(map)
984 val stopsShown = showStops(map)
985 val margin = pin.height
986 var selected: GeoPoint? = null
987 for ((id, position) in stops) {
988 if (id == selectedId) {
989 selected = position
990 continue
991 }
992 if (!stopsShown) continue
993 projection.toPixels(position, point)
994 if (point.x < -margin || point.y < -margin || point.x > map.width + margin || point.y > map.height + margin) continue
995 if (pins) {
996 // The pin's point sits on the stop
997 canvas.drawBitmap(pin, point.x - pin.width / 2f, point.y - pin.height.toFloat(), null)
998 } else {
999 canvas.drawCircle(point.x.toFloat(), point.y.toFloat(), dotRadius + dotRing, dotStroke)
1000 canvas.drawCircle(point.x.toFloat(), point.y.toFloat(), dotRadius, dotFill)
1001 }
1002 }
1003 selected?.let {
1004 projection.toPixels(it, point)
1005 canvas.drawBitmap(openPin, point.x - openPin.width / 2f, point.y - openPin.height.toFloat(), null)
1006 }
1007 }
1008
1009 override fun onSingleTapConfirmed(e: MotionEvent, map: MapView): Boolean {
1010 if (!showStops(map)) return false
1011 val projection = map.projection
1012 // A pin is tapped on its head, above the stop; a dot on the stop itself
1013 val lift = if (showPins(map)) pin.height * 0.6f else 0f
1014 var best: String? = null
1015 var bestDistance = tapRadius * tapRadius
1016 for ((id, position) in stops) {
1017 projection.toPixels(position, point)
1018 val dx = e.x - point.x
1019 val dy = e.y - (point.y - lift)
1020 val d = dx * dx + dy * dy
1021 if (d < bestDistance) {
1022 bestDistance = d
1023 best = id
1024 }
1025 }
1026 val id = best ?: return false
1027 onTap(id)
1028 return true
1029 }

1031 private companion object {
1032 /** From this zoom up the stops are pins; below it, dots */
1033 const val PIN_ZOOM = 15.0
1034
1035 /** Below this zoom the stops are not drawn at all */
1036 const val DOT_ZOOM = 13.5
1037 }
