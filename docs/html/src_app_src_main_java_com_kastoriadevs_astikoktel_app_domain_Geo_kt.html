14 /** Distance in metres between two points */
15 fun distanceMeters(aLat: Double, aLng: Double, bLat: Double, bLng: Double): Double {
16 val r = 6_371_000.0
17 val rad = Math.PI / 180
18 val dLat = (bLat - aLat) * rad
19 val dLng = (bLng - aLng) * rad
20 val h = sin(dLat / 2).pow(2) + cos(aLat * rad) * cos(bLat * rad) * sin(dLng / 2).pow(2)
21 return 2 * r * asin(sqrt(h))
22 }

30 fun decodePolyline(encoded: String): List<Pair<Double, Double>> {
31 val points = ArrayList<Pair<Double, Double>>()
32 var index = 0
33 var lat = 0
34 var lng = 0
35 fun next(): Int {
36 var result = 0
37 var shift = 0
38 var b: Int
39 do {
40 b = encoded[index++].code - 63
41 result = result or ((b and 0x1f) shl shift)
42 shift += 5
43 } while (b >= 0x20)
44 return if (result and 1 != 0) (result shr 1).inv() else result shr 1
45 }
46 while (index < encoded.length) {

48 lng += next()
