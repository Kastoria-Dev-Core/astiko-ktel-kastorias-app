19/**
20 * Reads text out loud with the phone's own text-to-speech engine, in the app's language.
21 * The engine starts in the background; whatever is asked before it's ready is said once it is.
22 */
23class Announcer(
24 context: Context,
25 private val locale: Locale,
26 private val onLanguageMissing: () -> Unit,
27) {
28 private lateinit var tts: TextToSpeech
29 private var ready = false
30 private var languageOk = true
31 private var pending: String? = null
32
33 init {
34 tts = TextToSpeech(context.applicationContext) { status -> onInit(status) }
35 }
36
37 private fun onInit(status: Int) {
38 if (status != TextToSpeech.SUCCESS) return
39 val result = tts.setLanguage(locale)
40 // Greek text read by an English voice is noise, so without a voice for the language we stay quiet
41 languageOk = result != TextToSpeech.LANG_MISSING_DATA && result != TextToSpeech.LANG_NOT_SUPPORTED
42 ready = true
43 pending?.let { say(it) }
44 pending = null
45 }
46
47 fun speak(text: String) {
48 if (ready) say(text) else pending = text
49 }
50
51 private fun say(text: String) {
52 if (!languageOk) {
53 onLanguageMissing()
54 return
55 }
56 tts.speak(text, TextToSpeech.QUEUE_FLUSH, null, UTTERANCE_ID)
57 }
58
59 fun stop() {
60 pending = null
61 if (ready) tts.stop()
62 }
63
64 fun shutdown() {
65 pending = null
66 tts.stop()
67 tts.shutdown()
68 }
69
70 private companion object {
71 const val UTTERANCE_ID = "astika-departures"
72 }
73}

75/** An [Announcer] for the current screen, released when the screen goes away */
76@Composable
77fun rememberAnnouncer(onLanguageMissing: () -> Unit): Announcer {
78 val context = LocalContext.current
79 val locale = LocalConfiguration.current.locales[0]
80 val announcer = remember(locale) { Announcer(context, locale, onLanguageMissing) }
81 DisposableEffect(announcer) { onDispose { announcer.shutdown() } }
82 return announcer
83}

85/**
86 * What is said for a stop: its name, then up to three departures, e.g.
87 * "Στάση Αφετηρία 2. Γραμμή 4, Πανεπιστήμιο, σε 14 λεπτά. Γραμμή 7, Εργατικές Κατοικίες, στις 16."
88 * Line numbers are said without the leading zero the badges show, and times in words a voice reads well.
89 */
90@Composable
91fun departuresAnnouncement(stopName: String, departures: List<Departure>, now: LocalDateTime): String {
92 if (departures.isEmpty()) return stringResource(R.string.voice_none, stopName)
93 val parts = departures.take(3).map { dep ->
94 val mins = Schedule.minutesUntil(dep, now)
95 val whenText = when {
96 dep.dayOffset > 0 -> stringResource(R.string.voice_tomorrow_at, spokenTime(dep.minutes))
97 mins <= 0 -> stringResource(R.string.voice_now)
98 mins < 60 -> pluralStringResource(R.plurals.voice_in_minutes, mins, mins)
99 else -> stringResource(R.string.voice_at, spokenTime(dep.minutes))
100 }
101 stringResource(R.string.voice_departure, dep.line.number.trimStart('0'), dep.line.name.text(), whenText)
102 }
103 return stringResource(R.string.voice_intro, stopName) + " " + parts.joinToString(" ")
104}

106@Composable
107private fun spokenTime(minutes: Int): String {
108 val m = ((minutes % 1440) + 1440) % 1440
109 val hour = m / 60
110 val minute = m % 60
111 return if (minute == 0) stringResource(R.string.voice_time_hour, hour) else stringResource(R.string.voice_time, hour, minute)
112}
