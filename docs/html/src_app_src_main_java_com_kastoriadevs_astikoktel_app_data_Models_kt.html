122 /** Lines that stop at a stop, in line order, including the stops of alternative endings */
123 fun linesAtStop(stopId: String): List<Line> =
124 lines.filter { line -> line.directions.any { d -> d.stops.any { it.stop == stopId } || d.branches.any { stopId in it.stops } } }

126 /** Every stop a line can call at, in either direction or on an alternative ending */

130 /** Stops of a direction in order, for drawing it on the map */
131 fun directionStops(line: Line, directionId: String): List<Stop> {
132 val direction = line.directions.firstOrNull { it.id == directionId } ?: line.directions.first()
133 return direction.stops.mapNotNull { stop(it.stop) }
134 }
135
136 /** The path a direction takes on the map as (lat, lng): along the streets when known, else stop to stop */
137 fun directionPath(line: Line, directionId: String): List<Pair<Double, Double>> {
138 val direction = line.directions.firstOrNull { it.id == directionId } ?: line.directions.first()
139 direction.shape?.let { return Geo.decodePolyline(it) }
140 return directionStops(line, directionId).map { it.lat to it.lng }
141 }
142
143 /** Paths of a direction's alternative endings */
144 fun branchPaths(line: Line, directionId: String): List<List<Pair<Double, Double>>> {
145 val direction = line.directions.firstOrNull { it.id == directionId } ?: return emptyList()
146 return direction.branches.map { branch ->
147 branch.shape?.let { Geo.decodePolyline(it) } ?: branch.stops.mapNotNull { stop(it) }.map { it.lat to it.lng }
148 }
149 }
