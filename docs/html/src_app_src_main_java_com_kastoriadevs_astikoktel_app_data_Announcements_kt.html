61/**
62 * Downloads the announcements feed and keeps the last good copy on disk, so the
63 * list still shows offline and the first frame doesn't wait for the network.
64 */
65class AnnouncementsRepository(
66 private val feedUrl: String,
67 private val cacheFile: File,
68 private val userAgent: String,
69) {
70 private val json = Json { ignoreUnknownKeys = true }
71 private val mutex = Mutex()
72 private var lastFetch: Instant = Instant.EPOCH
73
74 private val _announcements = MutableStateFlow<List<Announcement>?>(null)
75
76 /** null until the cache or the network has answered */
77 val announcements: StateFlow<List<Announcement>?> = _announcements.asStateFlow()
78
79 private val _failed = MutableStateFlow(false)
80
81 /** The last refresh couldn't reach the feed */
82 val failed: StateFlow<Boolean> = _failed.asStateFlow()
83
84 /** Fetches the feed unless it was fetched in the last [MIN_INTERVAL], or [force] */
85 suspend fun refresh(force: Boolean = false) = withContext(Dispatchers.IO) {
86 mutex.withLock {
87 if (_announcements.value == null) readCache()
88 if (!force && Duration.between(lastFetch, Instant.now()) < MIN_INTERVAL) return@withLock
89
90 try {
91 val body = download()
92 val feed = json.decodeFromString<AnnouncementFeed>(body)
93 _announcements.value = feed.announcements
94 cacheFile.writeText(body)
95 lastFetch = Instant.now()
96 _failed.value = false
97 } catch (e: IOException) {
98 _failed.value = true
99 } catch (e: SerializationException) {
100 _failed.value = true
101 }
102 // Show an empty list rather than loading forever when there's no cache either
103 if (_announcements.value == null) _announcements.value = emptyList()
104 }
105 }
106
107 private fun readCache() {
108 if (!cacheFile.exists()) return
109 try {
110 _announcements.value = json.decodeFromString<AnnouncementFeed>(cacheFile.readText()).announcements
111 } catch (e: Exception) {
112 cacheFile.delete()
113 }
114 }

116 private fun download(): String {
117 val connection = URL(feedUrl).openConnection() as HttpURLConnection
118 try {
119 connection.connectTimeout = 10_000
120 connection.readTimeout = 10_000
121 connection.setRequestProperty("Accept", "application/json")
122 connection.setRequestProperty("User-Agent", userAgent)
123 if (connection.responseCode != HttpURLConnection.HTTP_OK) {
124 throw IOException("HTTP ${connection.responseCode}")
125 }
126 return connection.inputStream.bufferedReader(Charsets.UTF_8).use { it.readText() }
127 } finally {
128 connection.disconnect()
129 }
130 }
