67fun StopDetailScreen(
68 stopId: String,
69 appVm: AppViewModel,
70 onBack: () -> Unit,
71 onOpenLine: (lineId: String, dirId: String?) -> Unit,
72 onOpenMap: (String) -> Unit,
73 onOpenAnnouncements: () -> Unit,
74 announcementsVm: AnnouncementsViewModel = viewModel(factory = AnnouncementsViewModel.Factory),
75) {
76 StatusBarIcons(darkIcons = !isAppDark())
77 val container = appContainer()
78 val data = container.transit.data
79 val schedule = container.schedule
80 val stop = data.stop(stopId) ?: return
81 val now = rememberNow()
82 val favorites by appVm.favorites.collectAsStateWithLifecycle()
83 val announcements by announcementsVm.announcements.collectAsStateWithLifecycle()
84 val lines = remember(stop.id) { data.linesAtStop(stop.id) }
85 var lineFilter by rememberSaveable { mutableStateOf<String?>(null) }
86
87 val departures = schedule.nextDepartures(stop.id, now, 40)
88 .filter { lineFilter == null || it.line.id == lineFilter }
89 .take(12)
90 val previous = schedule.previousDeparture(stop.id, now)?.takeIf { lineFilter == null || it.line.id == lineFilter }
91
92 // Voice announcements: on by default if Settings says so, and this stop's speaker button overrides it
93 val voiceDefault by appVm.voice.collectAsStateWithLifecycle()
94 var voiceChoice by rememberSaveable(stop.id) { mutableStateOf<Boolean?>(null) }
95 val voiceOn = voiceChoice ?: voiceDefault
96 val context = LocalContext.current
97 val voiceMissing = stringResource(R.string.voice_missing)
98 val announcer = rememberAnnouncer(onLanguageMissing = { Toast.makeText(context, voiceMissing, Toast.LENGTH_SHORT).show() })
99 val spoken = departures.take(3)
100 val announcement = departuresAnnouncement(stop.name.text(), spoken, now)
101 // Said when the stop opens or the voice is turned on, when the line filter changes, when the first bus
102 // leaves, and once more when the next one is two minutes away. Not every minute.
103 val nextIsClose = spoken.firstOrNull()?.let { it.dayOffset == 0 && Schedule.minutesUntil(it, now) <= 2 } ?: false
104 val spokenKey = spoken.map { "${it.line.id}-${it.direction.id}-${it.minutes}-${it.dayOffset}" }
105 LaunchedEffect(voiceOn, spokenKey, nextIsClose) {
106 if (voiceOn) announcer.speak(announcement) else announcer.stop()
107 }
108 // The feed doesn't say which lines a post is about, so the newest one still in force is shown
109 val notice = remember(announcements, now.toLocalDate()) {
110 announcements?.firstOrNull { it.isActive(now.toLocalDate(), Instant.now()) }
111 }

173 Text(
