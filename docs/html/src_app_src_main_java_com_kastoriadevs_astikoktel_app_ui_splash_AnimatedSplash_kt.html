61/*
62 * Animated splash, built from the app icon (B3).
63 * Design space: 390 x 844 units (same as the approved storyboard). The art sits at 39% of the height,
64 * the waves stick to the bottom edge. Timings in ms:
65 * 0-430 waves rise, 290-1220 route draws, stops pop at 320 and 720, bus at 1120, title at 1370,
66 * loader from 1550 until the data is ready, then a 360 ms exit that reveals Home underneath.
67 */
68
69private const val INTRO_MS = 1800f
70private const val EXIT_MS = 360
71
72private val EaseOut: Easing = CubicBezierEasing(0.2f, 0.8f, 0.2f, 1f)
73private val RouteEasing: Easing = CubicBezierEasing(0.65f, 0f, 0.35f, 1f)
74private val Overshoot: Easing = CubicBezierEasing(0.34f, 1.56f, 0.64f, 1f)

78 .toPath()
79private val Wave2: Path = PathParser()
80 .parsePathString("M0 770 C 110 738, 230 772, 390 716 V 844 H 0 Z")

93/** Progress 0..1 of [t] inside [start, end], eased */
94private fun window(t: Float, start: Float, end: Float, easing: Easing): Float =
95 easing.transform(((t - start) / (end - start)).coerceIn(0f, 1f))
96
97/** Stop pop: 0 -> 1.25 -> 1 */
98private fun pop(t: Float, start: Float): Float = when {
99 t <= start -> 0f
100 t < start + 180f -> 1.25f * EaseOut.transform((t - start) / 180f)
101 t < start + 288f -> 1.25f - 0.25f * ((t - start - 180f) / 108f)
102 else -> 1f
103}
104
105/** Small double hop of the bus after it lands, in icon units */
106private fun hop(t: Float): Float {
107 fun lerp(a: Float, b: Float, f: Float) = a + (b - a) * f.coerceIn(0f, 1f)
108 return when {
109 t < 1440f -> 0f
110 t < 1548f -> lerp(0f, -8f, (t - 1440f) / 108f)
111 t < 1656f -> lerp(-8f, 0f, (t - 1548f) / 108f)
112 t < 1728f -> lerp(0f, -3f, (t - 1656f) / 72f)
113 t < 1800f -> lerp(-3f, 0f, (t - 1728f) / 72f)
114 else -> 0f
115 }
116}

118/**
119 * Full-screen splash. Plays the intro once, keeps the loader running until [ready],
120 * then fades out (revealing what is composed underneath) and calls [onFinished].
121 */
122@Composable
123fun AnimatedSplash(ready: Boolean, onFinished: () -> Unit) {
124 StatusBarIcons(darkIcons = false)
125 val context = LocalContext.current
126 val reduceMotion = remember {
127 Settings.Global.getFloat(context.contentResolver, Settings.Global.ANIMATOR_DURATION_SCALE, 1f) == 0f
128 }
129
130 val time = remember { Animatable(if (reduceMotion) INTRO_MS else 0f) }
131 val exit = remember { Animatable(0f) }
132 val introDone by remember { derivedStateOf { time.value >= INTRO_MS } }
133
134 LaunchedEffect(Unit) {
135 if (!reduceMotion) time.animateTo(INTRO_MS, tween(INTRO_MS.toInt(), easing = LinearEasing))
136 }
137 LaunchedEffect(introDone, ready) {
138 if (introDone && ready) {
139 if (!reduceMotion) exit.animateTo(1f, tween(EXIT_MS, easing = FastOutSlowInEasing))
140 onFinished()
141 }
142 }
143
144 val loader = rememberInfiniteTransition(label = "splash-loader")
145 val barX by loader.animateFloat(
146 initialValue = -56f,
147 targetValue = 56f,
148 animationSpec = infiniteRepeatable(tween(640, easing = LinearEasing), RepeatMode.Restart),
149 label = "splash-loader-bar",
150 )
151
152 val t = time.value
153 val e = exit.value
154
155 BoxWithConstraints(
156 modifier = Modifier
157 .fillMaxSize()
158 .graphicsLayer { alpha = 1f - e }
159 .background(Lake)
160 // Swallow touches so nothing underneath reacts while the splash is up
161 .pointerInput(Unit) {
162 awaitPointerEventScope {
163 while (true) awaitPointerEvent().changes.forEach { it.consume() }
164 }
165 },
166 ) {
167 val unit = maxWidth / 390f
168 val artCenterY = maxHeight * 0.39f
169
170 Box(
171 modifier = Modifier
172 .fillMaxSize()
173 .graphicsLayer {
174 val s = 1f + 0.06f * e
175 scaleX = s
176 scaleY = s
177 transformOrigin = TransformOrigin(0.5f, 0.39f)
178 },
179 ) {
180 Canvas(modifier = Modifier.fillMaxSize()) {
181 val s = size.width / 390f
182 val bottom = size.height - 844f * s
183
184 val w1 = window(t, 0f, 432f, EaseOut)
185 if (w1 > 0f) {
186 withTransform({
187 translate(0f, bottom + (1f - w1) * 90f * s)
188 scale(s, s, pivot = Offset.Zero)
189 }) { drawPath(Wave1, LakeMid, alpha = w1) }
190 }
191 val w2 = window(t, 144f, 576f, EaseOut)

197 }

200 val cy = size.height * 0.39f
201 withTransform({

203 scale(s, s, pivot = Offset.Zero)

211 Column(
212 modifier = Modifier

256/** The icon art in icon units (512 space), at time [t] */
257private fun DrawScope.drawArt(t: Float) {
258 val route = window(t, 288f, 1224f, RouteEasing)
259 if (route > 0f) {
260 val measure = PathMeasure().apply { setPath(RoutePath, false) }
261 val segment = Path()
262 measure.getSegment(0f, RouteLength * route, segment, true)
263 drawPath(segment, RouteYellow, style = Stroke(width = 36f, cap = StrokeCap.Round))
264 }
265 drawStop(Stop1, pop(t, 324f))
266 drawStop(Stop2, pop(t, 720f))
267
268 val bus = window(t, 1116f, 1440f, Overshoot)
269 if (bus > 0f) {
270 withTransform({ scale(bus, bus, pivot = Bus) }) {
271 drawCircle(OffWhite, radius = 64f, center = Bus)
272 withTransform({ translate(0f, hop(t)) }) { drawBusGlyph() }
273 }
274 }
275}
