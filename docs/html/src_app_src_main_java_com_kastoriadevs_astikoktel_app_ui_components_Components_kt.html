192 * One departure. Within the hour the minutes are what you read first, with the clock time under
193 * them; later ones show the clock time only. A bus that already left is shown faded.
194 */
195@Composable
196fun DepartureRow(dep: Departure, now: LocalDateTime, onClick: () -> Unit, departed: Boolean = false) {
197 val mins = Schedule.minutesUntil(dep, now)
198 val clock = Schedule.formatMinutes(dep.minutes)
199 val within = !departed && dep.dayOffset == 0 && mins < 60
200 Row(
201 modifier = Modifier
202 .fillMaxWidth()
203 .heightIn(min = 60.dp)
204 .clickable(onClick = onClick)
205 .then(if (departed) Modifier.alpha(0.55f) else Modifier)
206 .padding(vertical = 8.dp),
207 verticalAlignment = Alignment.CenterVertically,
208 horizontalArrangement = Arrangement.spacedBy(12.dp),
209 ) {
210 LineBadge(dep.line)
211 // The line's name says which bus it is; the last stop only says which way it goes
212 Column(modifier = Modifier.weight(1f)) {
213 Text(
214 text = dep.line.name.text(),
215 fontSize = 15.sp,
216 fontWeight = FontWeight.Medium,
217 maxLines = 1,
218 overflow = TextOverflow.Ellipsis,
219 )
220 Row(verticalAlignment = Alignment.CenterVertically, horizontalArrangement = Arrangement.spacedBy(6.dp)) {
221 Text(
222 text = when {
223 departed -> stringResource(R.string.departed)
224 else -> stringResource(R.string.towards, dep.direction.headsign.text())
225 },
226 fontSize = 12.sp,
227 color = Astika.colors.muted,
228 maxLines = 1,
229 overflow = TextOverflow.Ellipsis,
230 modifier = Modifier.weight(1f, fill = false),
231 )
232 if (!departed && dep.last && dep.dayOffset == 0) Pill(stringResource(R.string.last_trip))
233 }
234 }
235 Column(horizontalAlignment = Alignment.End) {
236 when {
237 departed -> Text(clock, fontSize = 15.sp, fontWeight = FontWeight.Medium, color = Astika.colors.muted)
238 within -> {
239 Text(
240 text = if (mins <= 0) stringResource(R.string.now) else stringResource(R.string.minutes_only, mins),
241 fontSize = 22.sp,
242 fontWeight = FontWeight.SemiBold,
243 lineHeight = 24.sp,
244 )
245 Text(clock, fontSize = 12.sp, color = Astika.colors.muted)
246 }
247 else -> {
248 Text(clock, fontSize = 17.sp, fontWeight = FontWeight.SemiBold)
249 if (dep.dayOffset > 0) Text(stringResource(R.string.tomorrow), fontSize = 12.sp, color = Astika.colors.muted)
250 }
251 }
252 }
253 }

256/** Departures in one list; [previous] is the bus that just left, shown faded on top */
257@Composable
258fun DepartureList(
259 departures: List<Departure>,
260 now: LocalDateTime,
261 previous: Departure? = null,
262 onOpen: (Departure) -> Unit,
263) {
264 if (departures.isEmpty() && previous == null) {
265 Text(stringResource(R.string.no_service), color = Astika.colors.muted, fontSize = 14.sp)
266 return
267 }
268 Column {
269 if (previous != null) {
270 DepartureRow(previous, now, onClick = { onOpen(previous) }, departed = true)
271 }
272 departures.forEachIndexed { i, dep ->
273 if (i > 0 || previous != null) RowDivider()
274 DepartureRow(dep, now, onClick = { onOpen(dep) })
275 }
276 if (departures.isEmpty()) {
277 RowDivider()
278 Text(
279 stringResource(R.string.no_more_today),
280 color = Astika.colors.muted,
281 fontSize = 14.sp,
282 modifier = Modifier.padding(vertical = 12.dp),
283 )
284 }
285 }

380/** Segmented control; on a tinted header the selected option turns white */
381@Composable
382fun <T> Segmented(
383 options: List<Pair<T, String>>,
384 selected: T,
385 onSelect: (T) -> Unit,
386 tinted: Boolean = false,
387) {
388 val track = if (tinted) Color.Black.copy(alpha = 0.2f) else Astika.colors.track
389 Row(
390 modifier = Modifier
391 .fillMaxWidth()
392 .clip(RoundedCornerShape(12.dp))
393 .background(track)
394 .padding(4.dp),
395 horizontalArrangement = Arrangement.spacedBy(4.dp),
396 ) {
397 options.forEach { (value, label) ->
398 val on = value == selected
399 val bg = when {
400 !on -> Color.Transparent
401 tinted -> Color.White
402 else -> Astika.colors.segmentOn
403 }
404 val fg = when {
405 on && tinted -> Color(0xFF10212B)
406 on -> MaterialTheme.colorScheme.onSurface
407 tinted -> Color.White.copy(alpha = 0.92f)
408 else -> Astika.colors.ink2
409 }
410 SegmentOption(label, bg, fg, on, Modifier.weight(1f)) { onSelect(value) }
411 }
412 }
413}
