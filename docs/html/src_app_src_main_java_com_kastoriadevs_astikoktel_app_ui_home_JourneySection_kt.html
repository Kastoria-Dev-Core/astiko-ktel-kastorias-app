155@Composable
156private fun JourneyBody(
157 data: TransitData,
158 nearby: NearbyState,
159 plan: JourneyPlan?,
160 showAll: Boolean,
161 onToggleShowAll: () -> Unit,
162 onLocate: () -> Unit,
163 onOpenStop: (String) -> Unit,
164) {
165 if (nearby !is NearbyState.Ready || plan == null) {
166 val loading = nearby is NearbyState.Loading
167 JourneyMessage(
168 stringResource(if (loading) R.string.locating else R.string.journey_need_location),
169 actionText = if (loading) null else stringResource(R.string.find_nearby),
170 onAction = onLocate,
171 )
172 return
173 }
174
175 if (plan.status != JourneyStatus.AlreadyThere && plan.status != JourneyStatus.TooFar) {
176 val stop = plan.usefulStop ?: plan.nearestStop
177 StopRow(
178 stop = stop,
179 label = stringResource(if (plan.usefulStop != null) R.string.journey_useful_stop else R.string.journey_nearest_stop),
180 onClick = { onOpenStop(stop.stop.id) },
181 )
182 if (plan.usefulStop != null) {
183 Text(
184 stringResource(
185 R.string.journey_nearest_no_bus,
186 plan.nearestStop.stop.name.text(),
187 Geo.formatDistance(plan.nearestStop.distanceMeters, currentLanguage()),
188 ),
189 fontSize = 13.sp,
190 color = Astika.colors.muted,
191 modifier = Modifier.padding(start = 4.dp, top = 2.dp),
192 )
193 }
194 }
195
196 when (plan.status) {
197 JourneyStatus.TooFar -> JourneyMessage(stringResource(R.string.journey_too_far))
198 JourneyStatus.AlreadyThere -> JourneyMessage(stringResource(R.string.journey_already_there, plan.nearestStop.stop.name.text()))
199 JourneyStatus.NoRoute -> JourneyMessage(stringResource(R.string.journey_no_route))
200 JourneyStatus.NoMoreBuses -> JourneyMessage(stringResource(R.string.journey_no_more_buses))
201 JourneyStatus.Ready -> Column {
202 val shown = if (showAll) plan.options else plan.options.take(COLLAPSED_OPTIONS)
203 Column(
204 verticalArrangement = Arrangement.spacedBy(6.dp),
205 modifier = Modifier.padding(top = 10.dp),
206 ) {
207 shown.forEachIndexed { index, option ->
208 JourneyOptionRow(
209 option = option,
210 alightName = data.stop(option.ride.alightStopId)?.name?.text() ?: option.ride.alightStopId,
211 highlighted = index == 0,
212 onClick = { onOpenStop(option.ride.board.stop.id) },
213 )
214 }
215 }
216 if (plan.options.size > COLLAPSED_OPTIONS) {
217 TextButton(onClick = onToggleShowAll, modifier = Modifier.align(Alignment.CenterHorizontally)) {
218 Text(
219 if (showAll) stringResource(R.string.journey_show_less)
220 else stringResource(R.string.journey_show_all, plan.options.size),
221 color = Astika.colors.link,
222 fontWeight = FontWeight.SemiBold,
223 )
224 Icon(
225 if (showAll) Icons.Filled.KeyboardArrowUp else Icons.Filled.KeyboardArrowDown,
226 contentDescription = null,
227 tint = Astika.colors.link,
228 )
229 }
230 }
231 Text(
232 stringResource(R.string.journey_note),
233 fontSize = 12.sp,
234 color = Astika.colors.muted,
235 modifier = Modifier.padding(start = 4.dp, top = if (plan.options.size > COLLAPSED_OPTIONS) 0.dp else 8.dp),
236 )
237 }
238 }

328private fun JourneyOptionRow(option: JourneyOption, alightName: String, highlighted: Boolean, onClick: () -> Unit) {
329 val c = Astika.colors
330 val board = option.ride.board
331 val strong = if (highlighted) c.heroText else MaterialTheme.colorScheme.onSurface
332 val soft = if (highlighted) c.heroMuted else c.muted
333 val number = if (highlighted) c.heroText else c.link
334
335 Row(
336 verticalAlignment = Alignment.CenterVertically,
337 modifier = Modifier
338 .fillMaxWidth()
339 .clip(RoundedCornerShape(16.dp))
340 .background(if (highlighted) c.hero else MaterialTheme.colorScheme.surfaceVariant)
341 .clickable(onClick = onClick)
342 .padding(horizontal = 12.dp, vertical = 10.dp),
343 ) {
344 LineBadge(option.ride.line, BadgeSize.MD)
345 Column(
346 modifier = Modifier
347 .weight(1f)
348 .padding(horizontal = 10.dp),
349 ) {
350 if (highlighted) {
351 Text(stringResource(R.string.journey_fastest), fontSize = 12.sp, fontWeight = FontWeight.SemiBold, color = soft)
352 }
353 Text(
354 option.ride.line.name.text(),
355 fontSize = 15.sp,
356 fontWeight = FontWeight.Bold,
357 color = strong,
358 maxLines = 1,
359 overflow = TextOverflow.Ellipsis,
360 )
361 // "Get on: Καρυδιά 🚶18′": a long stop name gets "…", the walking time never does
362 Row(verticalAlignment = Alignment.CenterVertically) {
363 Text(
364 stringResource(R.string.journey_board, board.stop.title()),
365 fontSize = 13.sp,
366 color = soft,
367 maxLines = 1,
368 overflow = TextOverflow.Ellipsis,
369 modifier = Modifier.weight(1f, fill = false),
370 )
371 Icon(
372 painterResource(R.drawable.ic_walk),
373 contentDescription = null,
374 tint = soft,
375 modifier = Modifier
376 .padding(start = 6.dp)
377 .size(13.dp),
378 )
379 Text(stringResource(R.string.journey_walk_minutes, board.walkMinutes), fontSize = 13.sp, color = soft, maxLines = 1)
380 }
381 Text(
382 stringResource(R.string.journey_alight, alightName),
383 fontSize = 13.sp,
384 color = soft,
385 maxLines = 1,
386 overflow = TextOverflow.Ellipsis,
387 )
388 }
389 Column(horizontalAlignment = Alignment.CenterHorizontally) {
390 val big = if (highlighted) 28.sp else 22.sp
391 when {
392 option.minutesUntil == 0 -> Text(stringResource(R.string.now), color = strong, fontWeight = FontWeight.ExtraBold, fontSize = 20.sp)
393 // "368 min" is hard to read: an hour or more away we write "6h 8′"
394 option.minutesUntil > 59 -> {
395 val hours = option.minutesUntil / 60
396 val minutes = option.minutesUntil % 60
397 Text(
398 if (minutes == 0) stringResource(R.string.duration_hours, hours)
399 else stringResource(R.string.duration_hours_minutes, hours, minutes),
400 color = number,
401 fontWeight = FontWeight.ExtraBold,
402 fontSize = if (highlighted) 22.sp else 18.sp,
403 )
404 }
405 else -> {
406 Text(option.minutesUntil.toString(), color = number, fontWeight = FontWeight.ExtraBold, fontSize = big, lineHeight = big)
407 Text(stringResource(R.string.minutes_short), color = soft, fontSize = 11.sp)
408 }
409 }
410 Text(Schedule.formatMinutes(option.minutes), color = strong, fontWeight = FontWeight.SemiBold, fontSize = 12.sp)
411 option.later?.let {
412 Text(stringResource(R.string.journey_later, Schedule.formatMinutes(it)), color = soft, fontSize = 11.sp)
413 }
414 }
415 }
