218/** Within the hour: minutes, with the clock time under them. Later: the clock time. None left today: a dash. */
219@Composable
220private fun ArrivalLabel(arrival: Int?, nowMin: Int, isLastStop: Boolean) {
221 if (arrival == null) {
222 Text(
223 stringResource(R.string.none_expected),
224 fontSize = 12.sp,
225 color = Astika.colors.muted,
226 modifier = Modifier.padding(start = 8.dp),
227 )
228 return
229 }
230 val mins = arrival - nowMin
231 Column(horizontalAlignment = Alignment.End, modifier = Modifier.padding(start = 8.dp)) {
232 if (mins < 60) {
233 Text(
234 if (mins <= 0) stringResource(R.string.now) else stringResource(R.string.in_min, mins),
235 fontSize = 15.sp,
236 fontWeight = FontWeight.SemiBold,
237 )
238 Text(Schedule.formatMinutes(arrival), fontSize = 12.sp, color = Astika.colors.muted)
239 } else {
240 Text(
241 Schedule.formatMinutes(arrival),
242 fontSize = 15.sp,
243 fontWeight = if (isLastStop) FontWeight.Medium else FontWeight.Normal,
244 color = Astika.colors.ink2,
245 )
