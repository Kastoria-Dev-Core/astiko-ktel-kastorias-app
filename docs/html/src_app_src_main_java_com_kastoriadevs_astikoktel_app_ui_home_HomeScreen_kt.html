288/** What kind of day it is, in one word for the header; the line pages keep the timetable's full names */
289private fun dayLabelRes(schedule: com.kastoriadevs.astikoktel.app.domain.Schedule, date: java.time.LocalDate): Int = when {
290 schedule.isHoliday(date) -> R.string.day_holiday
291 else -> when (schedule.serviceDayFor(date)) {
292 ServiceDay.WEEKDAY -> R.string.day_weekday
293 ServiceDay.SATURDAY -> R.string.day_saturday
294 ServiceDay.SUNDAY -> R.string.day_sunday
295 }
296}

298@Composable
299private fun buildDateLine(now: java.time.LocalDateTime, dayRes: Int): String {
300 val locale = LocalConfiguration.current.locales[0]
301 val formatter = remember(locale) { DateTimeFormatter.ofPattern("EEEE d MMMM", locale) }
302 val date = now.format(formatter).replaceFirstChar { it.titlecase(locale) }
303 return "$date · ${stringResource(dayRes)}"
304}
