33/** Current time, refreshed every [intervalMs] while the screen is composed */
34@Composable
35fun rememberNow(intervalMs: Long = 15_000L): LocalDateTime {
36 var now by remember { mutableStateOf(LocalDateTime.now()) }
37 LaunchedEffect(intervalMs) {
38 while (true) {
39 delay(intervalMs)
40 now = LocalDateTime.now()
41 }
42 }
43 return now
44}

56/** A stop's name as shown where it's picked or opened: "Αφετηρία (αναχωρήσεις)" when it has a qualifier */
57@Composable
58fun Stop.title(): String {
59 val base = name.text()
60 val extra = qualifier?.text() ?: return base
61 // An English name that already ends in brackets, "Afetiria (Terminal)", gets "(Terminal, departures)"
62 return if (base.endsWith(")")) "${base.dropLast(1)}, $extra)" else "$base ($extra)"
